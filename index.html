Ponte Nativa Android TV / Chromecast 4 (WebView JavascriptInterface)

Para lançar pacotes nativos, listar apps instalados e abrir as Configurações do Sistema no Android TV / Chromecast 4, exponha AndroidLauncher no WebView: 

class TVBridge(private val ctx: Context) {
    @JavascriptInterface
    fun launchApp(packageName: String) {
        ctx.packageManager.getLaunchIntentForPackage(packageName)?.let {
            it.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            ctx.startActivity(it)
        }
    }
    @JavascriptInterface
    fun openSystemSettings() {
        val intent = Intent(Settings.ACTION_SETTINGS).apply {
            addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        }
        ctx.startActivity(intent)
    }
    @JavascriptInterface
    fun getInstalledApps(): String { /* retorna JSON [{name, packageName, iconBase64}] */ }
}
webView.addJavascriptInterface(TVBridge(this), "AndroidLauncher")
